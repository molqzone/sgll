cmake_minimum_required(VERSION 3.22)
project(sgll_regressions LANGUAGES C CXX)

set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/.." sgll)

# Run the same behavior checks through C and C++ consumers of the C library.
configure_file(registers.c registers_cpp.cpp COPYONLY)
add_executable(sgll_registers_c registers.c)
add_executable(sgll_registers_cpp "${CMAKE_CURRENT_BINARY_DIR}/registers_cpp.cpp")
target_link_libraries(sgll_registers_c PRIVATE sgll)
target_link_libraries(sgll_registers_cpp PRIVATE sgll)
configure_file(init.c init_cpp.cpp COPYONLY)
add_executable(sgll_init_c init.c)
add_executable(sgll_init_cpp "${CMAKE_CURRENT_BINARY_DIR}/init_cpp.cpp")
target_link_libraries(sgll_init_c PRIVATE sgll)
target_link_libraries(sgll_init_cpp PRIVATE sgll)
configure_file(mbox.c mbox_cpp.cpp COPYONLY)
add_executable(sgll_mbox_c mbox.c)
add_executable(sgll_mbox_cpp "${CMAKE_CURRENT_BINARY_DIR}/mbox_cpp.cpp")
target_link_libraries(sgll_mbox_c PRIVATE sgll)
target_link_libraries(sgll_mbox_cpp PRIVATE sgll)
configure_file(peripherals.c peripherals_cpp.cpp COPYONLY)
add_executable(sgll_peripherals_c peripherals.c)
add_executable(sgll_peripherals_cpp "${CMAKE_CURRENT_BINARY_DIR}/peripherals_cpp.cpp")
target_link_libraries(sgll_peripherals_c PRIVATE sgll)
target_link_libraries(sgll_peripherals_cpp PRIVATE sgll)

enable_testing()
add_test(NAME registers_c COMMAND sgll_registers_c)
add_test(NAME registers_cpp COMMAND sgll_registers_cpp)
add_test(NAME init_c COMMAND sgll_init_c)
add_test(NAME init_cpp COMMAND sgll_init_cpp)
add_test(NAME mbox_c COMMAND sgll_mbox_c)
add_test(NAME mbox_cpp COMMAND sgll_mbox_cpp)
add_test(NAME peripherals_c COMMAND sgll_peripherals_c)
add_test(NAME peripherals_cpp COMMAND sgll_peripherals_cpp)
